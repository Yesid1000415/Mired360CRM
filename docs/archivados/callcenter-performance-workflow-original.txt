function performanceState(sales,meta,assigned=true){
  if(!assigned)return{key:'gray',label:'Sin asesor'};
  const now=new Date(),days=new Date(now.getFullYear(),now.getMonth()+1,0).getDate();
  const expected=Math.max(1,meta*(now.getDate()/days));
  const ratio=sales/expected;
  if(sales>=meta||ratio>=1)return{key:'green',label:'Va cumpliendo'};
  if(ratio>=0.60)return{key:'yellow',label:'En progreso'};
  return{key:'red',label:'Necesita impulso'};
}
function monthStartIso(){const d=new Date();return new Date(d.getFullYear(),d.getMonth(),1,0,0,0).toISOString()}
async function loadCallCenterTeam(){
  const {data:exts,error}=await sb.from('callcenter_extensiones').select('*').order('extension');
  if(error){$('callCenterTeam').innerHTML='<div class="empty">No se pudo cargar el rendimiento: '+esc(error.message)+'</div>';return}
  callcenterExts=exts||[];
  const isCoord=['coordinador','admin'].includes(String(currentProfile?.rol||'').toLowerCase());
  if(isCoord){const {data}=await sb.from('perfiles').select('user_id,nombre,email,rol').eq('rol','asesor').order('nombre');advisorProfiles=data||[]}
  const visibleBases=bases||[],baseMap=new Map(visibleBases.map(b=>[b.id,b.extension]));
  let allContacts=[];
  if(visibleBases.length){const {data}=await sb.from('contactos_base_llamadas').select('base_id,estado,fecha_venta').in('base_id',visibleBases.map(b=>b.id));allContacts=data||[]}
  const start=monthStartIso();
  const html=callcenterExts.map(ext=>{
    const sales=allContacts.filter(c=>baseMap.get(c.base_id)===ext.extension&&c.estado==='Venta'&&c.fecha_venta&&c.fecha_venta>=start).length;
    const meta=Number(ext.meta_mensual||15),st=performanceState(sales,meta,!!ext.asesor_user_id);
    const options='<option value="">Sin asesor</option>'+advisorProfiles.map(a=>`<option value="${a.user_id}" ${a.user_id===ext.asesor_user_id?'selected':''}>${esc(a.nombre||a.email)}</option>`).join('');
    const admin=isCoord?`<div class="perf-admin"><select id="advisor-${ext.extension}">${options}</select><input id="meta-${ext.extension}" type="number" min="1" max="500" value="${meta}" aria-label="Meta mensual"><button class="btn ghost" onclick="saveExtensionAssignment('${ext.extension}')">Guardar asignación</button></div>`:'';
    return `<div class="perf-card perf-${st.key}"><div><span class="perf-dot dot-${st.key}"></span><b>Extensión ${extensionShort(ext.extension)}</b></div><div class="light">${esc(ext.asesor_nombre||'Sin asesor asignado')}</div><div class="big">${sales} <span class="light">ventas</span></div><div><b>${st.label}</b></div><div class="light">Meta mensual: ${meta}</div>${admin}</div>`;
  }).join('');
  $('callCenterTeam').innerHTML=html||'<div class="empty">No hay extensiones asignadas.</div>';
}
async function saveExtensionAssignment(extension){
  const sel=$('advisor-'+extension),metaEl=$('meta-'+extension);if(!sel||!metaEl)return;
  const userId=sel.value||null,meta=Math.max(1,Number(metaEl.value||15));
  const advisor=advisorProfiles.find(a=>a.user_id===userId)||null;
  const patch={asesor_user_id:userId,asesor_nombre:advisor?.nombre||null,meta_mensual:meta,actualizado_en:new Date().toISOString()};
  const {error}=await sb.from('callcenter_extensiones').update(patch).eq('extension',extension);if(error)return alert('No se pudo guardar la asignación. '+error.message);
  const {error:baseError}=await sb.from('bases_llamadas').update({asesor_user_id:userId,asesor_nombre:advisor?.nombre||null,actualizado_en:new Date().toISOString()}).eq('extension',extension);if(baseError)console.warn(baseError);
  await loadBases();await loadCallCenterTeam();alert(`Extensión ${extensionShort(extension)} actualizada correctamente.`);
}
